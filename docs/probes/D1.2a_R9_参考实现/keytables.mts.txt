const FR: Record<string, number> = { '½': 0.5, '¼': 0.25, '¾': 0.75 }
const toNum = (s0: string): number | null => { let s = untag(s0).replace(/[,，\s]/g, '').replace(/^[×x+]/, ''); if (s === '' || /^[—–-]+$/.test(s)) return null; for (const [k, v] of Object.entries(FR)) if (s.includes(k)) { const i = s.replace(k, ''); return (i ? const int = (s: string): number => { const n = toNum(s); if (n === null || !Number.isInteger(n)) throw new Error('整数解析失败 ' + s); return n }
const level = (s: string): number => { const m = /^(\d{1,2})(?:环|th|st|nd|rd|级)?$/.exec(untag(s).replace(/\s/g, '')); if (!m) throw new Error('等级解析失败 ' + s); return Number(m[1]) }
const levelNoun = (s: string): number => (/戏法/.test(s) ? 0 : level(s))
const range = (s0: string): { min: number; max: number | null } => { const s = untag(s0).replace(/\s/g, '').replace(/[–—]/g, '-').replace(/级$/, ''); let m = /^(\d+)-(\d+)$/.exec(s); if (m) return { min: Number(m[1]), max: Number(m[2]) }; m = /^(\d+)(?:或更高|及以上)$/.exec(s); if (m) return { min: Number(m[1]), max: null }; m = /^(\d+)$/.exec(s); if (m) return { min: Number(m[1]), max: Number(m[1]) }; throw new Error('区间解析失败 ' + s0) }
const dice = (s0: string): string => { const m = /(\d*)d(\d+)/i.exec(untag(s0)); if (!m) throw new Error('骰子解析失败 ' + s0); return `${m[1] === '1' ? '' : m[1]}d${m[2]}` }
const bonus = (s: string): number => { const m = /^([+-]?\d+)\s*$/.exec(untag(s)); if (!m) throw new Error('加值解析失败 ' + s); return Number(m[1]) }
const bonusRange = (s: string) => { const m = /^\+?(\d+)至\+?(\d+)$/.exec(untag(s).replace(/\s/g, '')); if (!m) throw new Error('加值区间解析失败 ' + s); return { min: Number(m[1]), max: Number(m[2]) } }
const SIZE: Record<string, string> = { 微型: 'T', 小型: 'S', 中型: 'M', 大型: 'L', 巨型: 'H', 超巨型: 'G' }
const size = (s: string): string => { const k = Object.keys(SIZE).sort((a, b) => b.length - a.length).find((x) => untag(s).startsWith(x)); if (!k) throw new Error('体型解析失败 ' + s); return SIZE[k] }
const crv = (s: string): { cr: string; value: number } => { const t = untag(s).replace(/\s/g, ''); const m = /^(\d+)(?:\/(\d+))?$/.exec(t); if (!m) throw new Error('CR 解析失败 ' + s); return { cr: t, value: m[2] ? Number(m[1]) / Number(m[2]) : Number(m[1]) } }
const diceAvg = (s: string) => { const t = untag(s); const m = /^(\d+)\s*\((.+)\)$/.exec(s.replace(/\{@dice ([^}|]*)[^}]*\}/g, '$1').trim()); if (!m) throw new Error('带平均骰解析失败 ' + s); return { average: Number(m[1]), dice: dice(m[2]) } }
